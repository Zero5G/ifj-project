<prog> -> <prolog> <function_list>
<prolog> -> KW_IMPORT ifj26 TT_DOT TT_STAR
<function_list> -> <function> <function_list>
<function_list> -> ε
<function> -> KW_DEF TT_IDENTIFIER <parameters> TT_COLON <return_type> <block>
<parameters> -> TT_LPAREN <parameter> TT_RPAREN
<parameter> -> TT_IDENTIFIER TT_COLON <type>
<parameter> -> ε
<return_type> -> <type>
<type> -> KW_INT
<type> -> KW_DOUBLE
<type> -> KW_STRING
<type> -> KW_BOOLEAN
<type> -> KW_UNIT
<type> -> KW_LIST
<block> -> TT_ASSIGN <statement>
<block> -> TT_ASSIGN TT_EOL TT_INDENT <body> TT_OUTDENT
<body> -> <statement_list>
<statement_list> -> <statement> <statement_list>
<statement_list> -> ε
<statement> -> <variable_definition>
<statement> -> <assignment>
<statement> -> <function_call>
<statement> -> <function_return>
<statement> -> <if_statement>
<statement> -> <while_statement>
<statement> -> <expression>
