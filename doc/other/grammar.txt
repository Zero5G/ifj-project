<prog> -> <prolog> <function_list>
<prolog> -> KW_IMPORT ifj26 TT_DOT TT_STAR
<function_list> -> <function> <function_list>
<function_list> -> ε
<function> -> KW_DEF TT_IDENTIFIER TT_LPAREN <parameters> TT_RPAREN <return_type> <block>
<parameters> -> <parameter> <parameter_next>
<parameters> -> ε
<parameter> -> TT_IDENTIFIER TT_COLON <type>
<parameter_next> -> TT_COMMA TT_IDENTIFIER TT_COLON <type> <parameter_next>
<parameter_next> -> ε
<return_type> -> TT_COLON <type>
<return_type> -> ε
<type> -> KW_INT
<type> -> KW_DOUBLE
<type> -> KW_STRING
<type> -> KW_BOOLEAN
<type> -> KW_UNIT
<type> -> KW_LIST
<block> -> TT_ASSIGN <statement>
<block> -> TT_ASSIGN TT_EOL TT_INDENT <body> TT_OUTDENT
<body> -> <statement_list>
<statement_list> -> <statement> <statement_list>
<statement_list> -> ε
<statement> -> <variable_definition>
<statement> -> <assignment>
<statement> -> <function_call>
<statement> -> <function_return>
<statement> -> <if_statement>
<statement> -> <while_statement>
<statement> -> <expression>
<variable_definition> -> KW_VAL TT_IDENTIFIER <variable_type> TT_ASSIGN <expression>
<variable_definition> -> KW_VAR TT_IDENTIFIER <variable_type> TT_ASSIGN <expression>
<variable_type> -> TT_COLON <type>
<variable_type> -> ε
<assignment> -> TT_IDENTIFIER TT_ASSIGN <expression>
<function_call> -> TT_IDENTIFIER TT_LPAREN <argument_list> TT_RPAREN
<function_return> -> KW_RETURN
<if_statement> -> KW_IF <condition> KW_THEN TT_EOL TT_INDENT <body> TT_EOL TT_OUTDENT KW_ELSE TT_EOL TT_INDENT <body> TT_EOL TT_OUTDENT
<while_statement> -> KW_WHILE <condition> KW_DO TT_EOL TT_INDENT <body> TT_EOL TT_OUTDENT
